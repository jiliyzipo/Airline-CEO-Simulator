  760|         {
  761|             message = "";
  762|             if (gs == null || mail == null) { message = "邮件无效。"; return false; }
  763|             var permit = FindPermitMail(gs, mail.origin, mail.dest);
  764|             if (permit != null)
  765|             {
  766|                 permit.permitStage = RoutePermitLogic.StageNone;
  767|                 permit.permitGate = "";
  768|                 permit.permitNote = "已放式";
  769|                 permit.effectPending = false;
  770|             }
  771|             Mark(gs, mail.id, "rejected");
  772|             message = $"已放式 {mail.origin}-{mail.dest} 的准入申请，许可办理终止（日后可重新召集研究）。";
  773|             gs.AddDecisionLog("准入：放弹 " + mail.origin + "-" + mail.dest + " 航线经营许可申请");
  774|             gs.NotifyChanged();
  775|             return true;
  776|         }

  778|         /// <summary>承载准入许可的「办理单」＝对应航线的开航提案邮件。</summary>
  779|         public static InboxMail FindPermitMail(GameState gs, string origin, string dest)
  780|         {
  781|             if (gs?.inbox == null) return null;
  782|             for (int i = 0; i < gs.inbox.Count; i++)
  783|             {
  784|                 var m = gs.inbox[i];
  785|                 if (m == null || m.action != "open_route") continue;
  786|                 if (m.permitStage < RoutePermitLogic.StageReview) continue;
  787|                 if (string.Equals(m.origin, origin, StringComparison.OrdinalIgnoreCase)
  788|                     && string.Equals(m.dest, dest, StringComparison.OrdinalIgnoreCase))
  789|                     return m;
  790|             }
  791|             return null;
  792|         }

  794|         /// <summary>把许可复核季度顺建 n 季（并保持待办）。</summary>
  795|         static void DeferPermit(InboxMail m, int quarters)
  796|         {
  797|             int ey = m.permitDueYear > 0 ? m.permitDueYear : m.effectYear;
  798|             int eq = m.permitDueQuarter > 0 ? m.permitDueQuarter : m.effectQuarter;
  799|             for (int i = 0; i < quarters; i++)
  800|                 NextQuarter(ey, eq, out ey, out eq);
  801|             m.permitDueYear = ey;
  802|             m.permitDueQuarter = eq;
  803|             m.effectYear = ey;
  804|             m.effectQuarter = eq;
  805|             m.effectPending = true;
  806|         }
